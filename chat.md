from pydantic import BaseModel
from datetime import datetime


class TradePlan(BaseModel):
    symbol: str
    entry: float
    stop: float
    target: float
    position_size: float
    expiry_days: int
    created_at: datetime

    @property
    def reward_to_risk(self) -> float:
        reward = self.target - self.entry
        risk = self.entry - self.stop
        if risk <= 0:
            return 0
        return reward / risk

import pandas as pd
import numpy as np
from state import LiquiditySwingState


class LiquiditySwingStateBuilder:

    def __init__(
        self,
        lookback: int = 30,
        equal_tolerance_pct: float = 0.4,
        compression_window: int = 10,
    ):
        self.lookback = lookback
        self.equal_tolerance_pct = equal_tolerance_pct
        self.compression_window = compression_window

    def build(self, df: pd.DataFrame) -> LiquiditySwingState:
        df = df.copy()
        recent = df.tail(self.lookback)

        current = recent.iloc[-1]
        prev = recent.iloc[-2]

        # -----------------------------
        # 1️⃣ Equal Low Detection
        # -----------------------------
        lows = recent["low"].values
        base_low = np.min(lows)

        tolerance = base_low * self.equal_tolerance_pct / 100

        equal_lows = [
            low for low in lows
            if abs(low - base_low) <= tolerance
        ]

        equal_low_count = len(equal_lows)
        equal_low_level = base_low

        # -----------------------------
        # 2️⃣ Compression Detection
        # -----------------------------
        recent_range = recent["high"] - recent["low"]
        range_std_recent = recent_range.tail(self.compression_window).std()
        range_std_past = (df["high"] - df["low"]).std()

        is_compressed = range_std_recent < range_std_past

        # -----------------------------
        # 3️⃣ Sweep Detection
        # -----------------------------
        sweep_detected = (
            current["low"] < equal_low_level
            and current["close"] > equal_low_level
        )

        # -----------------------------
        # 4️⃣ Reclaim Confirmation
        # -----------------------------
        reclaim_confirmed = (
            prev["low"] < equal_low_level
            and current["close"] > prev["high"]
        )

        # -----------------------------
        # 5️⃣ Volume Spike
        # -----------------------------
        avg_volume = recent["volume"].mean()
        sweep_volume_ratio = (
            current["volume"] / avg_volume if avg_volume > 0 else 0
        )

        # -----------------------------
        # 6️⃣ Range High
        # -----------------------------
        range_high = recent["high"].max()

        return LiquiditySwingState(
            equal_low_count=equal_low_count,
            equal_low_level=equal_low_level,
            is_compressed=is_compressed,
            sweep_detected=sweep_detected,
            reclaim_confirmed=reclaim_confirmed,
            sweep_volume_ratio=sweep_volume_ratio,
            current_price=current["close"],
            range_high=range_high,
        )

from datetime import datetime
from typing import Optional
from models import TradePlan
from state import LiquiditySwingState


class LiquiditySwingStrategy:

    def __init__(self, capital: float, risk_pct: float = 1.0):
        self.capital = capital
        self.risk_pct = risk_pct

    def generate_trade(
        self,
        symbol: str,
        state: LiquiditySwingState,
    ) -> Optional[TradePlan]:

        if not state.is_tradeable():
            return None

        entry = state.current_price
        stop = state.equal_low_level * 0.995  # slight buffer
        target = state.range_high

        risk_amount = self.capital * (self.risk_pct / 100)
        stop_distance = entry - stop

        if stop_distance <= 0:
            return None

        position_size = risk_amount / stop_distance

        trade = TradePlan(
            symbol=symbol,
            entry=entry,
            stop=stop,
            target=target,
            position_size=position_size,
            expiry_days=5,
            created_at=datetime.utcnow(),
        )

        if trade.reward_to_risk < 2:
            return None

        return trade
